# Migraciones de esquema. Sustituye a `create_all`, que solo crea tablas
# nuevas y no aplica cambios a las que ya tienen datos.
alembic==1.19.2

# Tokens JWT. Se elige PyJWT sobre `python-jose` (el de la documentación
# antigua de FastAPI): python-jose tiene mantenimiento irregular e historial
# de CVEs. PyJWT es Python puro, sin compilación ni dependencias pesadas.
PyJWT==2.13.0

# Tratamiento de imágenes. Valida que un archivo subido sea REALMENTE una
# imagen, abriéndolo, en vez de fiarse de la cabecera Content-Type que envía
# el cliente. También dibuja las siluetas del catálogo de ejemplo.
Pillow==12.3.0

# Cálculo sobre imágenes: el motor textil trabaja con matrices de píxeles, y en
# Python puro una sola prueba tardaría minutos en vez de medio segundo. Aquí
# hace el trabajo de verdad, no es una comodidad.
numpy==2.5.3

# Cliente oficial de la API de OpenAI. Solo se usa con AI_PROVIDER=openai, pero
# se instala siempre para que el import no dependa de la configuración.
#
# CUIDADO: genera imágenes de pago. El valor por defecto es AI_PROVIDER=none
# precisamente para que clonar el repositorio no empiece a gastar dinero.
openai==3.15.0

# Visión por computador para el recorte de la prenda y la veta de la tela:
# GrabCut (corte de grafo con modelos de color) para separar prenda y fondo
# donde un umbral no basta —una camiseta blanca sobre gris claro, el resplandor
# del fondo de estudio alrededor de una cazadora—, transformada de distancia y
# componentes conexas.
#
# La variante HEADLESS: sin interfaz gráfica, que en un servidor sobra y trae
# dependencias de escritorio. Rama 4 y no la 5, que es un salto mayor y reciente.
opencv-python-headless==4.14.0.94

# --- El probador: la persona de la foto con la prenda puesta ---
#
# Cliente oficial de los Spaces de Gradio. Llama a FASHN VTON 1.5 en su Space
# gratuito de Hugging Face (ver app/probador/fashn.py). Riesgo asumido: el
# Space es de un tercero y puede cambiar su API o caerse; el error se traduce
# a un mensaje, nunca a un fallo mudo.
gradio_client==2.7.1

# Descarga, una vez y con la revisión fijada, el analizador de personas. Ya lo
# trae gradio_client; se fija aquí porque el código lo usa directamente.
huggingface-hub==1.33.0

# Ejecuta el analizador de personas (SegFormer B2 para ropa, ONNX, 28 MB) en
# CPU: 0,8 s por foto sin tarjeta gráfica. Es lo que permite devolver a la
# foto original todo lo que el modelo cambia fuera de la prenda. Se eligió
# frente a PyTorch, que ocupa del orden de un gigabyte para hacer lo mismo.
onnxruntime==1.30.0
